# Import your libraries
import pyspark
from pyspark.sql.functions import *
from pyspark.sql.window import Window 

# Start writing code
res = (
    amazon_transactions
    .withColumn(
        'first_purchase',
        min('created_at').over(Window.partitionBy('user_id'))
    )
    .filter(
        (col('created_at') != col('first_purchase')) & 
        ( datediff('created_at', 'first_purchase') < 8 )
    )
    .select('user_id')
    .dropDuplicates()
)

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
