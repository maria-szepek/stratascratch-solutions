# Import your libraries
import pyspark
from pyspark.sql.functions import *
from pyspark.sql.window import Window
# Start writing code
res = (
    ms_download_facts.alias('d')
    .join(
        ms_user_dimension.alias('u'),
        col('d.user_id') == col('u.user_id'),
        'inner'
    )
    .join(
        ms_acc_dimension.alias('a'),
        col('u.acc_id') == col('a.acc_id'),
        'inner'
    )
    .withColumn(
        'daily_downloads_p',
        sum(when(col('paying_customer') == 'yes', col('downloads'))).over(Window.partitionBy('date'))
    )
    .withColumn(
        'daily_downloads_np',
        sum(when(col('paying_customer') == 'no', col('downloads'))).over(Window.partitionBy('date'))
    )
    .filter(
        col('daily_downloads_p') < col('daily_downloads_np')
    )
    .select(
        col('date'),
        col('daily_downloads_np'),
        col('daily_downloads_p')
    )
    .dropDuplicates()
    .orderBy(
        col('date')
    )
)

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
