# Import your libraries
import pyspark
from pyspark.sql.functions import *
from pyspark.sql.window import Window

# Start writing code
res = (
    customers.alias('c')
    .join(
        orders.alias('o'),
        customers['id'] == orders['cust_id'],
        'inner'
    )
    .filter(
        (col('o.order_date').between('2019-02-01', '2019-05-01'))
    )
    .groupBy(col('c.id'), col('c.first_name'), col('o.order_date'))
    .agg(
        sum('total_order_cost').alias('total_order_value')
    )
    .withColumn(
        'order_value_rank',
        dense_rank().over(
            Window.partitionBy(col('o.order_date')).orderBy(col('total_order_value').desc())
        )
    )
    .filter(
        col('order_value_rank') == 1
    )
    .select(
        col('first_name'),
        col('total_order_value'),
        col('order_date')
    )
    .orderBy(
        col('order_date')
    )
)


# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
