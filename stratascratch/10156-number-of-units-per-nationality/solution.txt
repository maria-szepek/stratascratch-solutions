# Import your libraries
import pyspark
from pyspark.sql.functions import *

# Start writing code
res = (
    airbnb_units.alias('u')
    .join(
        airbnb_hosts.alias('h'),
        col('u.host_id') == col('h.host_id'),
        'inner'
    )
    .filter(
        (col('u.unit_type') == 'Apartment') & (col('h.age') < 30)
    )
    .groupBy(
        col('h.nationality')
    )
    .agg(
        count_distinct(col('u.unit_id')).alias('count')
    )
    
)


# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
