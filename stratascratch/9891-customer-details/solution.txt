# Import your libraries
import pyspark
from pyspark.sql.functions import *

# Start writing code
res = (
    customers
    .join(
        orders,
        customers['id'] == orders['cust_id'],
        'left'
    )
    .select(
        col('first_name'),
        col('last_name'),
        col('city'),
        col('order_details')
    )
    .orderBy(col('first_name').asc())
)

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
