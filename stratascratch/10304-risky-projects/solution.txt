# Import your libraries
import pyspark
from pyspark.sql.functions import * 

# Start writing code
res = (
    linkedin_projects.alias('p')
    .join(
        linkedin_emp_projects.alias('ep'),
        col('p.id') == col('ep.project_id'),
        'inner'
    )
    .join(
        linkedin_employees.alias('e'),
        col('ep.emp_id') == col('e.id'),
        'inner'
    )
    .withColumn(
        'project_employee_salary',
        ( date_diff(col('p.end_date'),col('p.start_date')) / 365) * col('e.salary') #  + 1
    )
    .groupBy(
        col('p.title'),
        col('p.budget')
    )
    .agg(
        sum('project_employee_salary').alias('total_employee_expenses')
        # ceil(sum('project_employee_salary')).alias('total_employee_expenses')
    )
    .filter(
        col('budget') < col('total_employee_expenses')
    )
    .withColumn(
        'total_employee_expenses',
        ceil(col('total_employee_expenses'))
    )
)





# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
