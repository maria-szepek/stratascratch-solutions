# Import your libraries
import pyspark
from pyspark.sql.functions import *
# Start writing code
res = (
    customers
    .filter(
        col('first_name').isin('Jill', 'Eva')
    )
    .join(
        orders,
        customers['id'] == orders['cust_id'],
        'inner'
    )
    .select(
        col('order_date'),
        col('order_details'),
        col('total_order_cost'),
        col('first_name')
    )
    .orderBy(
        col('cust_id')
    )
)

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
