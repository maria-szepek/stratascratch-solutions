# Import your libraries
import pyspark
from pyspark.sql.functions import *
from pyspark.sql.window import Window

# Start writing code
res = ms_employee_salary.withColumn(
    'current_salary', row_number().over(Window.partitionBy('id').orderBy(col('salary').desc()))
).filter(
    col('current_salary') == 1
).select(
    "id",
    "first_name",
    "last_name",
    "department_id",
    "salary"
).orderBy(
    col('id').asc()
)


# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
