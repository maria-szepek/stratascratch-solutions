# Import your libraries
import pyspark
from pyspark.sql.functions import *

# Start writing code
res = facebook_reactions.filter(
    col('reaction') == 'heart'
).join(
    facebook_posts, facebook_reactions["post_id"] == facebook_posts["post_id"], "inner"
).select(
    facebook_posts["post_id"],
    facebook_posts["poster"],
    facebook_posts["post_text"],
    facebook_posts["post_keywords"],
    facebook_posts["post_date"]
).dropDuplicates()
# .show()

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
