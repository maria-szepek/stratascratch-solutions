# Import your libraries
import pyspark
from pyspark.sql.functions import *
from pyspark.sql.window import Window

# Start writing code
res = (
    google_gmail_emails
    .groupBy(
        col('from_user')
    )
    .agg(
        count('id').alias('count_emails_sent')
    )
    .withColumn(
        'rank',
        row_number().over(Window.orderBy(col('count_emails_sent').desc(), col('from_user')))
    )
    .orderBy(
        col('rank')
    )
)

# To validate your solution, convert your final PySpark df to a pandas df
res.toPandas()
